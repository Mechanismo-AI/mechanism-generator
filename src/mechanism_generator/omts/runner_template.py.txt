#!/usr/bin/env python3
"""Run a prepared plan with the installed engine and verified model files.

This helper is portable and invokes Python directly without a shell. The chosen
Python environment must already contain the research engine's dependencies.
"""
import argparse
import hashlib
from importlib.resources import files
import json
from pathlib import Path
import subprocess
import sys


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--models-directory", type=Path, required=True)
    parser.add_argument("--check-only", action="store_true")
    args = parser.parse_args()
    base = Path(__file__).resolve().parent
    models = args.models_directory.resolve()
    plan = json.loads((base / "run_plan.json").read_text(encoding="utf-8"))
    engine = files("mechanism_generator.engine")
    for name, expected in plan["expected_engine_artifacts"].items():
        artifact = engine.joinpath(name)
        if not artifact.is_file() or hashlib.sha256(artifact.read_bytes()).hexdigest() != expected:
            parser.error(f"Missing or mismatched engine artifact: {name}; regenerate the plan after upgrading.")
    for name, expected in plan["expected_artifacts"].items():
        artifact = models / name
        if not artifact.is_file() or hashlib.sha256(artifact.read_bytes()).hexdigest() != expected:
            parser.error(f"Missing or mismatched model artifact: {name}")
    if args.check_only:
        print("Engine and model hashes match; the plan has not been executed.")
        return
    for run in plan["runs"]:
        command = [sys.executable, "-m", "mechanism_generator.engine", "--models-directory", str(models), *run["arguments"],
                   "--targets_file", str(base / run["targets_file"]),
                   "--output_root", str(base / run["output_root"])]
        subprocess.run(command, cwd=base, check=True)


if __name__ == "__main__":
    main()
