
# --------------------------------------------------------------- main
def main(argv):
    """Entry point: two full report builds compared for byte
    identity (G5), the scorecard printout, then the fail-closed
    gates. Exit 0: every internal check passed (engineering
    verdicts are data). Exit 1: an internal witness failed."""
    import argparse
    ap = argparse.ArgumentParser(
        description="dualcam-lever-transfer-06 deterministic "
                    "design script (Option B: cam + linkage lever)")
    ap.add_argument("--task",
                    default=("jobs/sixbar-gap-transfer-02/"
                             "gap_transfer_nostops.motion.json"),
                    help="frozen -02 task file (read-only)")
    args = ap.parse_args(argv)
    rep1 = build_report(args.task)
    rep2 = build_report(args.task)
    det = ("PASS (two-audit byte identity)"
           if json.dumps(rep1, sort_keys=True)
           == json.dumps(rep2, sort_keys=True) else "FAIL")
    rep1["determinism"] = det
    print_report(rep1)

    task_ok = rep1["task"]["ok"]
    carry_ok = rep1["cam_y"]["carry_ok"]
    lv = rep1["lever"]
    if lv.get("feasible") and lv.get("pick") is not None:
        lever_ok = (
            max(lv["xc_err"], lv["vc_err"]) <= 1e-9
            and max(lv["stop_v"]) <= 1e-12
            and lv["contact_x"]["min_local"] >= ROLLER_R - 0.01
            and rep1["contact_y"]["min_local"] >= ROLLER_R - 0.01
            and lv["contact_x"]["coarse_min"] >= ROLLER_R - 0.01
            and rep1["contact_y"]["coarse_min"] >= ROLLER_R - 0.01
            and lv["contact_x"]["alpha_dev"] <= 0.1
            and rep1["contact_y"]["rho_xval_rel"] <= 1e-9)
    else:
        # an infeasible design is recorded data, never an error exit
        lever_ok = True
    if not task_ok or not carry_ok or not lever_ok \
            or not det.startswith("PASS"):
        print("FAIL-CLOSED: an internal witness failed (exit 1)")
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
